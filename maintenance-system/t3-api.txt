<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Concerns\RegistaDescargas;
use App\Http\Controllers\Controller;
use App\Models\Discharge;
use App\Models\DischargeOperation;
use Carbon\Carbon;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class DischargeApiController extends Controller
{
    use RegistaDescargas;

    public function operations()
    {
        return DischargeOperation::open()
            ->where('status', 'active')
            ->latest()->get()
            ->map(fn ($op) => $this->operationPayload($op))
            ->values();
    }

    public function operation(DischargeOperation $operation)
    {
        return $this->operationPayload($operation);
    }

    public function store(Request $request)
    {
        $request->validate([
            'operation_id'     => 'required|exists:discharge_operations,id',
            'data'             => 'required|date',
            'motorista'        => 'required|string|max:100',
            'matricula'        => 'required|string|max:20',
            'transportadora'   => 'required|string|max:100',
            'numero_guia'      => 'required|string|unique:descargas,numero_guia',
            'hora_saida_porto' => 'nullable|date_format:H:i',
            'tipo_atrelado'    => 'required|in:normal,link',
            'items'            => 'required|array',
            'items.*'          => 'nullable|numeric|min:0',
            'assinatura'       => ['required', 'string', 'max:700000', function ($attr, $value, $fail) {
                $prefixo = 'data:image/png;base64,';
                $bin = str_starts_with($value, $prefixo) ? base64_decode(substr($value, strlen($prefixo)), true) : false;
                if ($bin === false || ! str_starts_with($bin, "\x89PNG")) {
                    $fail('A assinatura do motorista é inválida. Peça-lhe para assinar de novo.');
                }
            }],
        ], [
            'assinatura.required' => 'A assinatura do motorista é obrigatória.',
        ]);

        $operation = DischargeOperation::findOrFail($request->operation_id);
        abort_unless($operation->isActive(), 403, 'A operação não está em curso.');

        [$items, $qtys] = $this->quantidades($request, $operation, null);

        $discharge = DB::transaction(function () use ($request, $operation, $items, $qtys) {
            $transp = $this->resolverTransportadora($operation, $request->transportadora);

            $discharge = Discharge::create([
                'operation_id'     => $operation->id,
                'data'             => $request->data,
                'motorista'        => $request->motorista,
                'matricula'        => $request->matricula,
                'transportadora'   => $transp,
                'tipo_atrelado'    => $request->tipo_atrelado,
                'numero_guia'      => $request->numero_guia,
                'hora_saida_porto' => $request->hora_saida_porto,
                'registado_por'    => auth()->id(),
                'status'           => 'in_transit',
            ] + $this->totais($items, $qtys));

            $this->gravarItens($discharge, $qtys);

            $discharge->signature()->create([
                'nome'         => $request->motorista,
                'imagem'       => $request->assinatura,
                'assinado_por' => auth()->id(),
            ]);

            return $discharge;
        });

        return response()->json([
            'id'      => $discharge->id,
            'message' => 'Descarga registada. Aguarda confirmação na balança.',
        ], 201);
    }

    private function operationPayload(DischargeOperation $op): array
    {
        $camioes   = $op->discharges()->count();
        $pendentes = $op->discharges()->whereIn('status', ['pending', 'in_transit'])->count();
        $recebido  = (float) $op->discharges()->sum('peso_liquido');

        $ts = $op->discharges()
            ->whereIn('status', ['pending', 'in_transit'])
            ->whereNotNull('hora_saida_porto')
            ->get()
            ->map(fn ($d) => Carbon::parse($d->data->format('Y-m-d') . ' ' . substr((string) $d->hora_saida_porto, 0, 5))->timestamp)
            ->min();

        return [
            'id'              => $op->id,
            'navio'           => $op->navio,
            'empresa'         => $op->empresa,
            'bl'              => $op->bl,
            'factura'         => $op->factura,
            'data_chegada'    => $op->data_chegada ? $op->data_chegada->format('d/m/Y') : null,
            'status'          => $op->status,
            'esperado_ton'    => (float) $op->quantidade_esperada,
            'recebido_ton'    => round($recebido / 1000, 3),
            'camioes_count'   => $camioes,
            'pendentes_count' => $pendentes,
            'pendente_desde'  => $ts ? $ts * 1000 : null,
            'transportadoras' => collect($op->transportadoras ?? [])->pluck('nome')->values(),
            'locais'          => $op->locations()->get()
                ->sortByDesc(fn ($l) => (int) $l->pivot->principal)
                ->map(fn ($l) => [
                    'id'        => $l->id,
                    'nome'      => $l->nome,
                    'tipo'      => $l->tipo,
                    'principal' => (bool) $l->pivot->principal,
                ])->values(),
            'items'           => $this->itensComSaldo($op, null)->map(fn ($it) => [
                'id'               => $it->id,
                'nome'             => $it->nome,
                'unidade'          => $it->unidade,
                'unidade_label'    => $it->unidade_label,
                'peso_unitario_kg' => (float) $it->peso_unitario_kg,
                'plano'            => (float) $it->quantidade,
                'registado'        => (float) ($it->registado ?? 0),
                'saldo'            => (float) $it->quantidade - (float) ($it->registado ?? 0),
            ])->values(),
        ];
    }
}

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\Discharge;

class DischargeShowApiController extends Controller
{
    public function show(Discharge $discharge)
    {
        $discharge->load(['operation', 'items.operationItem', 'registeredBy', 'confirmedBy', 'signature']);

        $op     = $discharge->operation;
        $extras = $op ? $op->transportadorasExtra() : [];
        $sig    = $discharge->signature;

        $itens = $discharge->items->map(function ($l) {
            $it = $l->operationItem;
            $kg = $it->unidade === 'tonelada' ? 1000 : (float) $it->peso_unitario_kg;

            return [
                'id'                => $l->id,
                'nome'              => $it->nome,
                'unidade'           => $it->unidade,
                'unidade_label'     => $it->unidade_label,
                'quantidade'        => (float) $l->quantidade,
                'kg_unidade'        => $kg,
                'peso_kg'           => (float) $l->quantidade * $kg,
                'quantidade_confirmada' => $l->quantidade_confirmada !== null ? (float) $l->quantidade_confirmada : null,
            ];
        })->values();

        $balanca = null;
        if ($discharge->peso_bruto !== null || $discharge->tara !== null) {
            $balanca = [
                'hora_chegada' => $discharge->hora_chegada_balanca ? substr((string) $discharge->hora_chegada_balanca, 0, 5) : null,
                'peso_bruto'   => $discharge->peso_bruto !== null ? (float) $discharge->peso_bruto : null,
                'tara'         => $discharge->tara !== null ? (float) $discharge->tara : null,
                'peso_liquido' => $discharge->peso_liquido !== null ? (float) $discharge->peso_liquido : null,
                'confirmado_por' => $discharge->confirmedBy?->name,
                'divergencia'  => $discharge->hasDivergence(),
            ];
        }

        return [
            'id'                  => $discharge->id,
            'operation_id'        => $discharge->operation_id,
            'navio'               => $op?->navio,
            'status'              => $discharge->status,
            'data'                => $discharge->data->format('d/m/Y'),
            'numero_guia'         => $discharge->numero_guia,
            'hora_saida_porto'    => $discharge->hora_saida_porto ? substr((string) $discharge->hora_saida_porto, 0, 5) : null,
            'registado_por'       => $discharge->registeredBy?->name,
            'motorista'           => $discharge->motorista,
            'matricula'           => $discharge->matricula,
            'transportadora'      => $discharge->transportadora,
            'transportadora_extra'=> in_array(mb_strtolower((string) $discharge->transportadora), $extras, true),
            'tipo_atrelado'       => $discharge->tipo_atrelado,
            'atrelado_aviso'      => $discharge->atreladoInconsistente()
                                        ? 'não bate com os ' . (int) $discharge->total_sacos . ' sacos'
                                        : null,
            'assinatura'          => $sig ? [
                'imagem' => $sig->imagem,
                'nome'   => $sig->nome,
                'em'     => $sig->created_at?->format('d/m/Y H:i'),
            ] : null,
            'items'               => $itens,
            'total_sacos'         => (int) $discharge->total_sacos,
            'peso_total_estimado' => (float) $discharge->peso_total_estimado,
            'balanca'             => $balanca,
        ];
    }
}

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\DischargeOperation;
use Illuminate\Http\Request;

class DischargeListApiController extends Controller
{
    public function index(Request $request, DischargeOperation $operation)
    {
        $q = $operation->discharges();

        if ($s = trim((string) $request->query('q'))) {
            $like = '%' . mb_strtolower($s) . '%';
            $q->where(function ($w) use ($like) {
                $w->whereRaw('LOWER(matricula) LIKE ?', [$like])
                  ->orWhereRaw('LOWER(motorista) LIKE ?', [$like])
                  ->orWhereRaw('LOWER(numero_guia) LIKE ?', [$like])
                  ->orWhereRaw('LOWER(transportadora) LIKE ?', [$like]);
            });
        }

        $st = (string) $request->query('st');
        if ($st === 'por_confirmar') {
            $q->whereIn('status', ['pending', 'in_transit']);
        } elseif (in_array($st, ['pending', 'in_transit', 'weighing', 'confirmed'], true)) {
            $q->where('status', $st);
        }

        if ($tr = trim((string) $request->query('tr'))) {
            $q->whereRaw('LOWER(transportadora) = ?', [mb_strtolower($tr)]);
        }

        $at = (string) $request->query('at');
        if ($at === 'nd') {
            $q->whereNull('tipo_atrelado');
        } elseif (in_array($at, ['normal', 'link'], true)) {
            $q->where('tipo_atrelado', $at);
        }

        if ($de = $request->query('de'))   { $q->whereDate('data', '>=', $de); }
        if ($ate = $request->query('ate')) { $q->whereDate('data', '<=', $ate); }

        $porPagina = in_array((int) $request->query('pp'), [15, 30, 50], true) ? (int) $request->query('pp') : 15;

        $p = $q->orderByDesc('id')->paginate($porPagina);

        $data = $p->getCollection()->map(fn ($d) => [
            'id'                  => $d->id,
            'data'                => $d->data->format('Y-m-d'),
            'hora_saida_porto'    => $d->hora_saida_porto ? substr((string) $d->hora_saida_porto, 0, 5) : null,
            'motorista'           => $d->motorista,
            'matricula'           => $d->matricula,
            'transportadora'      => $d->transportadora,
            'numero_guia'         => $d->numero_guia,
            'tipo_atrelado'       => $d->tipo_atrelado,
            'status'              => $d->status,
            'total_sacos'         => (int) $d->total_sacos,
            'peso_total_estimado' => (float) $d->peso_total_estimado,
        ])->values();

        $porEstado = $operation->discharges()
            ->selectRaw('status, COUNT(*) as n')
            ->groupBy('status')
            ->pluck('n', 'status');

        return [
            'data' => $data,
            'meta' => [
                'pagina'    => $p->currentPage(),
                'ultima'    => $p->lastPage(),
                'por_pagina'=> $p->perPage(),
                'total'     => $p->total(),
                'de'        => $p->firstItem(),
                'ate'       => $p->lastItem(),
            ],
            'contagens' => [
                'todos'         => (int) $porEstado->sum(),
                'por_confirmar' => (int) (($porEstado['pending'] ?? 0) + ($porEstado['in_transit'] ?? 0)),
                'weighing'      => (int) ($porEstado['weighing'] ?? 0),
                'confirmed'     => (int) ($porEstado['confirmed'] ?? 0),
            ],
            'transportadoras' => $operation->discharges()
                ->select('transportadora')->distinct()->orderBy('transportadora')
                ->pluck('transportadora')->values(),
        ];
    }
}

<?php

namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Models\User;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;
use Illuminate\Validation\ValidationException;

class AuthApiController extends Controller
{
    public function login(Request $request)
    {
        $request->validate([
            'email'       => 'required|email',
            'password'    => 'required|string',
            'device_name' => 'nullable|string|max:60',
        ]);

        $user = User::where('email', $request->email)->first();

        if (! $user || ! Hash::check($request->password, $user->password)) {
            throw ValidationException::withMessages(['email' => 'Credenciais inválidas.']);
        }

        return [
            'token' => $user->createToken($request->device_name ?: 'tablet')->plainTextToken,
            'user'  => ['id' => $user->id, 'name' => $user->name, 'roles' => $user->getRoleNames()],
        ];
    }

    public function logout(Request $request)
    {
        $request->user()->currentAccessToken()->delete();
        return response()->noContent();
    }
}
